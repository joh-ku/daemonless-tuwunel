## Deployment

### Podman Compose

```yaml
services:
  daemonless-tuwunel:
    image: "ghcr.io/joh-ku/daemonless-tuwunel:latest"
    container_name: daemonless-tuwunel
    environment:
      - PUID=1000
      - PGID=1000
      - TZ=UTC
      - TUWUNEL_SERVER_NAME=matrix.example.org  # Homeserver server name (e.g. matrix.example.org)
      - TUWUNEL_ALLOW_REGISTRATION=false  # Allow public user registration (true or false)
    volumes:
      - "/containers/daemonless-tuwunel:/config"
      - "/containers/daemonless-tuwunel/data:/data"
    ports:
      - "8008:8008"
    # always (not unless-stopped) so FreeBSD's podman rc.d auto-starts it at boot
    restart: always
```

Save as `compose.yaml`, then run `podman-compose up -d`.

**appjail-director.yml**:

```yaml
# appjail-director.yml

options:
  - virtualnet: ':<random> default'
  - nat:
services:
  tuwunel:
    name: tuwunel
    options:
      - container: 'args:--pull'
      - expose: '8008:8008 proto:tcp'
    oci:
      user: root
      environment:
        - PUID: !ENV '${PUID}'
        - PGID: !ENV '${PGID}'
        - TZ: !ENV '${TZ}'
        - TUWUNEL_SERVER_NAME: !ENV '${TUWUNEL_SERVER_NAME}'
        - TUWUNEL_ALLOW_REGISTRATION: !ENV '${TUWUNEL_ALLOW_REGISTRATION}'
    volumes:
      - daemonless-tuwunel: /config
      - daemonless-tuwunel_data: /data
volumes:
  daemonless-tuwunel:
    device: '/containers/daemonless-tuwunel'
  daemonless-tuwunel_data:
    device: '/containers/daemonless-tuwunel/data'
```

**Makejail**:

```
# Makejail

ARG tag=latest

OPTION container=boot
OPTION overwrite=force
OPTION from=ghcr.io/joh-ku/daemonless-tuwunel:${tag}
```

Save the files above, then run `appjail-director up`.


> [!WARNING]
> Exposing ports in AppJail means that your service can be reached from remote hosts. If that is not your intention, do not expose the ports and communicate with the service using the jail's IPv4 address or hostname assigned by the virtual network.
>
> To avoid exposing ports, just remove the `expose` option in your `appjail-director.yml` or from your command-line arguments.
